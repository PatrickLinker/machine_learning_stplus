import os
import sys
import argparse
import subprocess

def run_command(command):
    """Führt einen Befehl plattformunabhängig aus."""
    # Verwende sys.executable, um sicherzustellen, dass dasselbe Python 
    # genutzt wird (wichtig für virtuelle Umgebungen)
    full_command = [sys.executable] + command
    print(f"Führe aus: {' '.join(full_command)}")
    
    # Der Parameter shell=True/False wird hier umgangen, indem wir eine Liste übergeben
    result = subprocess.run(full_command, capture_output=False, text=True)
    if result.returncode != 0:
        print(f"Fehler beim Ausführen von: {command}")
        sys.exit(result.returncode)

def main():
    parser = argparse.ArgumentParser(description="Zentrales Steuerungs-Skript für das Subset-Sum-Projekt")
    parser.add_argument("action", choices=["generate_data", "train", "evaluate", "all"], 
                        help="Die auszuführende Aktion")
    
    args = parser.parse_args()

    # Pfade plattformunabhängig mit os.path.join zusammenbauen
    script_dir = os.path.dirname(os.path.abspath(__file__))
    src_dir = os.path.join(script_dir, "src")

    if args.action == "generate_data" or args.action == "all":
        print("--- Starte Datengenerierung ---")
        run_command([os.path.join(src_dir, "dataset.py"), "--num_samples", "10000"])

    if args.action == "train" or args.action == "all":
        print("--- Starte Training ---")
        run_command([os.path.join(src_dir, "train.py"), "--epochs", "20"])

    if args.action == "evaluate" or args.action == "all":
        print("--- Starte Evaluation ---")
        run_command([os.path.join(src_dir, "predict.py")])

if __name__ == "__main__":
    main()
