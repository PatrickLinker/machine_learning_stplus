import os
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader
import numpy as np
from tqdm import tqdm

# Importiere unsere eigenen Module aus dem src-Ordner
from dataset import SubsetSumDataset
from model import SubsetSumDeepSets

def train_model(epochs=20, batch_size=64, lr=0.001, data_dir="data", model_dir="models"):
    # 1. Ordner für das fertige Modell erstellen
    os.makedirs(model_dir, exist_ok=True)
    
    # 2. Einlesen der binären .npy Dateien
    print("Lade binäre Datensätze...")
    S_data = np.load(os.path.join(data_dir, "S_data.npy"))
    T_data = np.load(os.path.join(data_dir, "T_data.npy"))
    labels = np.load(os.path.join(data_dir, "labels.npy"))
    
    # Datensatz aufteilen (80% Training, 20% Validierung)
    num_samples = len(labels)
    split_idx = int(num_samples * 0.8)
    
    train_dataset = SubsetSumDataset(S_data[:split_idx], T_data[:split_idx], labels[:split_idx])
    val_dataset = SubsetSumDataset(S_data[split_idx:], T_data[split_idx:], labels[split_idx:])
    
    # PyTorch DataLoader initialisieren
    train_loader = DataLoader(train_dataset, batch_size=batch_size, shuffle=True)
    val_loader = DataLoader(val_dataset, batch_size=batch_size, shuffle=False)
    
    # Hardware-Beschleunigung aktivieren falls vorhanden (Cuda/GPU)
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    print(f"Nutze Hardware-Beschleuniger: {device}")
    
    # 3. Modell, Loss und Optimizer initialisieren
    model = SubsetSumDeepSets().to(device)
    criterion = nn.BCELoss()
    optimizer = optim.Adam(model.parameters(), lr=lr)
    
    # 4. Der eigentliche Trainings-Loop
    print("Starte Training...")
    for epoch in range(1, epochs + 1):
        model.train()
        running_loss = 0.0
        correct_predictions = 0
        total_predictions = 0
        
        # tqdm fügt einen eleganten Ladebalken im Terminal/Colab hinzu
        progress_bar = tqdm(train_loader, desc=f"Epoche {epoch}/{epochs}")
        for S_batch, T_batch, label_batch in progress_bar:
            S_batch = S_batch.to(device)
            T_batch = T_batch.to(device)
            label_batch = label_batch.to(device)
            
            # Gradienten zurücksetzen
            optimizer.zero_grad()
            
            # Forward Pass (Vorhersage)
            predictions = model(S_batch, T_batch)
            loss = criterion(predictions, label_batch)
            
            # Backward Pass & Optimierung
            loss.backward()
            optimizer.step()
            
            # Metriken berechnen
            running_loss += loss.item() * S_batch.size(0)
            predicted_classes = (predictions >= 0.5).float() # Auf- oder Abrunden nach Beschreibung.docx
            correct_predictions += (predicted_classes == label_batch).sum().item()
            total_predictions += label_batch.size(0)
            
            # Ladebalken-Anzeige live aktualisieren
            progress_bar.set_postfix(loss=loss.item())
            
        epoch_loss = running_loss / total_predictions
        epoch_acc = (correct_predictions / total_predictions) * 100
        
        # 5. Validierungs-Schritt (Testen auf ungesehenen Daten)
        model.eval()
        val_correct = 0
        val_total = 0
        with torch.no_grad():
            for S_batch, T_batch, label_batch in val_loader:
                S_batch = S_batch.to(device)
                T_batch = T_batch.to(device)
                label_batch = label_batch.to(device)
                
                predictions = model(S_batch, T_batch)
                predicted_classes = (predictions >= 0.5).float()
                val_correct += (predicted_classes == label_batch).sum().item()
                val_total += label_batch.size(0)
                
        val_acc = (val_correct / val_total) * 100
        print(f"-> Ergebnis Epoche {epoch}: Train-Loss: {epoch_loss:.4f} | Train-Acc: {epoch_acc:.2f}% | Val-Acc: {val_acc:.2f}%\\n")
        
    # Modell-Gewichte abspeichern
    model_path = os.path.join(model_dir, "subset_sum_deepsets.pth")
    torch.save(model.state_dict(), model_path)
    print(f"Modell erfolgreich trainiert und gespeichert unter: {model_path}")

if __name__ == "__main__":
    import argparse
    parser = argparse.ArgumentParser()
    parser.add_argument("--epochs", type=int, default=15)
    parser.add_argument("--batch_size", type=int, default=64)
    parser.add_argument("--lr", type=float, default=0.001)
    args = parser.parse_args()
    
    train_model(epochs=args.epochs, batch_size=args.batch_size, lr=args.lr)
